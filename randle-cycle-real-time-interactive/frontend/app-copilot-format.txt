// ------------------------------------------------------------
// Flux color gradient helper
// ------------------------------------------------------------

function fluxColor(f) {

    const clamped =
        Math.min(f, 1.0);

    const r =
        Math.floor(
            255 * (1 - clamped)
        );

    const g = 255;

    const b =
        Math.floor(
            255 * (1 - clamped)
        );

    return (r << 16)
         + (g << 8)
         + b;
}

// ------------------------------------------------------------
// Inhibition color helper
// ------------------------------------------------------------

function inhibitionColor(x) {

    const c = Math.min(x, 1.0);

    const r = 255;
    const g = Math.floor(100 * (1 - c));
    const b = Math.floor(100 * (1 - c));

    return (r << 16) + (g << 8) + b;
}

// ------------------------------------------------------------
// Concentration color helper
// ------------------------------------------------------------

function concentrationColor(c) {

    const x = Math.min(
        Math.log10(1 + 10 * c),
        1.0
    );

    let r, g, b;

    if (x < 0.5) {

        // Cyan → Yellow

        const t = x / 0.5;

        r = Math.floor(255 * t);
        g = 255;
        b = Math.floor(255 * (1 - t));

    } else {

        // Yellow → Orange

        const t = (x - 0.5) / 0.5;

        r = 255;
        g = Math.floor(255 - 90 * t);
        b = 0;
    }

    return (r << 16) + (g << 8) + b;
}

// ------------------------------------------------------------
// Create PixiJS application
// ------------------------------------------------------------

const app = new PIXI.Application({

    width: 1400,
    height: 650,
    backgroundColor: 0x1e1e1e,

});

document.body.appendChild(app.view);

const fluxLayer = new PIXI.Container();
const inhibitionLayer = new PIXI.Container();
const nodeLayer = new PIXI.Container();
const labelLayer = new PIXI.Container();

app.stage.addChild(fluxLayer);
app.stage.addChild(inhibitionLayer);
app.stage.addChild(nodeLayer);
app.stage.addChild(labelLayer);

// ------------------------------------------------------------
// Containers
// ------------------------------------------------------------

let nodes = {};
let fluxEdges = [];
let inhibitionEdges = [];

const enzymeNodes = new Set([
    "GLUT4",
    "PGI",
    "PFK",
    "PDH",
    "ACL_ACC",
    "FAS",
    "CD36",
    "CPT1",
    "BetaOx"
]);

// ------------------------------------------------------------
// Create nodes
// ------------------------------------------------------------

function createNodes(nodePositions) {

    nodes = {};

    for (const [name, [x, y]] of Object.entries(nodePositions)) {

        const g = new PIXI.Graphics();

        g.beginFill(0x00aaff);
        g.drawCircle(0, 0, 20);
        g.endFill();

        g.x = x;
        g.y = y;

        const nameLabel = new PIXI.Text("", {
            fontFamily: "Arial",
            fontSize: 14,
            fill: 0xffffff
        });

        const concLabel = new PIXI.Text("", {
            fontFamily: "Arial",
            fontSize: 12,
            fill: 0xffffff
        });

        const rateLabel = new PIXI.Text("", {
            fontFamily: "Arial",
            fontSize: 10,
            fill: 0xcccccc
        });

        nodes[name] = {

            gfx: g,

            nameLabel: nameLabel,
            concLabel: concLabel,
            rateLabel: rateLabel

        };

        nodeLayer.addChild(g);
        labelLayer.addChild(nameLabel);
        labelLayer.addChild(concLabel);
        labelLayer.addChild(rateLabel);
    }
}

// ------------------------------------------------------------
// Create edges (flux and inhibition are separate)
// ------------------------------------------------------------

function createFluxEdges(edgeList) {

    fluxEdges = [];

    for (const [src, dst] of edgeList) {

        const line = new PIXI.Graphics();

        const s = nodes[src];
        const d = nodes[dst];

        line.lineStyle(2, 0xffffff, 0.5);

        line.moveTo(
            s.gfx.x,
            s.gfx.y
        );

        line.lineTo(
            d.gfx.x,
            d.gfx.y
        );

        fluxEdges.push({
            src,
            dst,
            gfx: line
        });

        fluxLayer.addChild(line);
    }
}

function createInhibitionEdges(edgeList) {

    inhibitionEdges = [];

    for (const [src, dst] of edgeList) {

        const line = new PIXI.Graphics();

        const s = nodes[src];
        const d = nodes[dst];

        line.lineStyle(2, 0x550000, 0.5);

        line.moveTo(
            s.gfx.x,
            s.gfx.y
        );

        line.lineTo(
            d.gfx.x,
            d.gfx.y
        );

        inhibitionEdges.push({
            src,
            dst,
            gfx: line
        });

        inhibitionLayer.addChild(line);
    }
}

// ------------------------------------------------------------
// Flux mapping
// ------------------------------------------------------------

const fluxMap = {

    // Glucose branch

    "Glc_ext→GLUT4": "GLUT4",
    "GLUT4→Glc": "GLUT4",

    "Glc→Glc6P": "Glycolysis",

    "Glc6P→PGI": "PGI",
    "PGI→Fru6P": "PGI",

    "Fru6P→PFK": "Glycolysis",
    "PFK→Fru16P2": "Glycolysis",

    "Fru16P2→Pyruvate": "Glycolysis",

    "Pyruvate→PDH": "PDH",
    "PDH→AcCoA": "PDH",

    // Citrate / lipogenesis branch

    "AcCoA→Cit": "Citrate_prod",

    "Cit→ACL_ACC": "ACL_ACC",
    "ACL_ACC→Mal": "ACL_ACC",

    "Mal→FAS": "FAS",
    "FAS→LCFA_CoA_cyto": "FAS",

    // Fat uptake branch

    "LCFA_ext→CD36": "CD36",
    "CD36→LCFA_CoA_cyto": "CD36",

    "LCFA_CoA_cyto→CPT1": "CPT1",
    "CPT1→LCFA_CoA_mito": "CPT1",

    "LCFA_CoA_mito→BetaOx": "BetaOx",
    "BetaOx→AcCoA": "BetaOx",

    // Storage

    "LCFA_CoA_cyto→TAG": "TAG"
};

const inhibitionMap = {

    "Cit→GLUT4": "GLUT4",

    "Cit→PFK": "PFK",

    "AcCoA→PDH": "PDH",

    "Mal→CPT1": "CPT1"
};

// ------------------------------------------------------------
// WebSocket
// ------------------------------------------------------------

const ws = new WebSocket("ws://localhost:8765");

const gluSlider = document.getElementById("gluSlider");
const fatSlider = document.getElementById("fatSlider");
const atpSlider = document.getElementById("atpSlider");

function sendInputs() {

    const msg = {

        glu_in: parseFloat(gluSlider.value),
        fat_in: parseFloat(fatSlider.value),
        atp_draw: parseFloat(atpSlider.value)

    };

    ws.send(JSON.stringify(msg));
}

gluSlider.oninput = sendInputs;
fatSlider.oninput = sendInputs;
atpSlider.oninput = sendInputs;

// ------------------------------------------------------------
// Receive updates
// ------------------------------------------------------------

ws.onmessage = (event) => {

    const data = JSON.parse(event.data);

    const state = data.state || {};
    const dstate = data.dstate || {};
    const flux = data.flux || {};
    const inh = data.inhibition || {};

    // --------------------------------------------------------
    // Update nodes
    // --------------------------------------------------------

    for (const [name, nodeObj] of Object.entries(nodes)) {

        const conc = state[name] || 0.0;
        
        const rate = dstate[name] || 0.0;

        const incoming =
            fluxEdges.filter(
                e => e.dst === name
            );

        let f_total = 0.0;

        for (const edge of incoming) {

            const key =
                `${edge.src}→${edge.dst}`;

            const fluxKey =
                fluxMap[key];

            if (fluxKey) {

                f_total +=
                    flux[fluxKey] || 0.0;
            }
        }

        const amplified =
            Math.min(
                f_total * 2.5,
                1.0
            );

        const inhVal =
            inh[name] || 0.0;

        const shrink =
            1 -
            0.3 *
            Math.min(inhVal, 1.0);

        const concScale =
            1 +
            0.5 *
            Math.log10(
                1 + 10 * conc
            );

        if (enzymeNodes.has(name)) {

            nodeObj.gfx.clear();

            nodeObj.gfx.beginFill(0xcc8844);

            nodeObj.gfx.drawRoundedRect(
                -30,
                -12,
                60,
                24,
                4
            );

            nodeObj.gfx.endFill();

            nodeObj.gfx.scale.set(1.0);

            nodeObj.nameLabel.anchor.set(0.5);

            nodeObj.nameLabel.text = name;
            nodeObj.concLabel.text = "";
            nodeObj.rateLabel.text = "";

            nodeObj.nameLabel.x = nodeObj.gfx.x;
            nodeObj.nameLabel.y = nodeObj.gfx.y;

            nodeObj.concLabel.x = nodeObj.gfx.x;
            nodeObj.concLabel.y = nodeObj.gfx.y;

            nodeObj.rateLabel.x = nodeObj.gfx.x;
            nodeObj.rateLabel.y = nodeObj.gfx.y;

        } else {

            nodeObj.gfx.clear();

            nodeObj.gfx.beginFill(
                concentrationColor(conc)
            );

            nodeObj.gfx.drawCircle(
                0,
                0,
                20
            );

            nodeObj.gfx.endFill();

            nodeObj.gfx.scale.set(
                concScale * shrink
            );

            nodeObj.nameLabel.anchor.set(0.5);
            nodeObj.concLabel.anchor.set(0.5);
            nodeObj.rateLabel.anchor.set(0.5);

            nodeObj.nameLabel.text = name;

            nodeObj.concLabel.text =
                conc.toFixed(3);

            nodeObj.rateLabel.text =
                rate.toExponential(2);

            nodeObj.nameLabel.x = nodeObj.gfx.x;
            nodeObj.nameLabel.y = nodeObj.gfx.y - 12;

            nodeObj.concLabel.x = nodeObj.gfx.x;
            nodeObj.concLabel.y = nodeObj.gfx.y + 28;

            nodeObj.rateLabel.x = nodeObj.gfx.x;
            nodeObj.rateLabel.y = nodeObj.gfx.y + 42;

        }
    }

    // --------------------------------------------------------
    // Update edges
    // --------------------------------------------------------

    for (const edge of fluxEdges) {
    
        const key =
            `${edge.src}→${edge.dst}`;
    
        const fluxKey =
            fluxMap[key];
    
        const f =
            fluxKey
                ? flux[fluxKey] || 0.0
                : 0.0;
    
        const amplified =
            Math.min(f * 3.0, 1.0);
    
        edge.gfx.clear();
    
        edge.gfx.lineStyle(
            2 + 10 * amplified,
            fluxColor(amplified),
            0.9
        );
    
        edge.gfx.moveTo(
            nodes[edge.src].gfx.x,
            nodes[edge.src].gfx.y
        );
    
        edge.gfx.lineTo(
            nodes[edge.dst].gfx.x,
            nodes[edge.dst].gfx.y
        );
    }


    for (const edge of inhibitionEdges) {
    
        const key =
            `${edge.src}→${edge.dst}`;
    
        const inhKey =
            inhibitionMap[key];
    
        const inhVal =
            inhKey
                ? inh[inhKey] || 0.0
                : 0.0;
    
        edge.gfx.clear();
    
        edge.gfx.lineStyle(
            2 + 10 * inhVal,
            inhibitionColor(inhVal),
            0.9
        );
    
        edge.gfx.moveTo(
            nodes[edge.src].gfx.x,
            nodes[edge.src].gfx.y
        );
    
        edge.gfx.lineTo(
            nodes[edge.dst].gfx.x,
            nodes[edge.dst].gfx.y
        );
    }
    
};
